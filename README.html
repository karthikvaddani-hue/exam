8(a) Add animations to UI elements using Flutter's animation framework

Question: Add animations to UI elements using Flutter's animation framework.

Program:

import "package:flutter/material.dart";
import "package:device_preview/device_preview.dart";

void main() {
  runApp(DevicePreview(builder: (_) => MyApp()));
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        appBar: AppBar(title: Text("Fade Animation")),
        body: FadeAnimation(),
      ),
    );
  }
}

class FadeAnimation extends StatefulWidget {
  @override
  _FadeAnimationState createState() => _FadeAnimationState();
}

class _FadeAnimationState extends State<FadeAnimation>
    with SingleTickerProviderStateMixin {

  late AnimationController _controller;
  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();

    _controller = AnimationController(
      vsync: this,
      duration: Duration(seconds: 2),
    );

    _animation =
        Tween<double>(begin: 0.0, end: 1.0).animate(_controller);

    _controller.forward();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Center(
      child: FadeTransition(
        opacity: _animation,
        child: Container(
          width: 200,
          height: 200,
          color: Colors.blue,
        ),
      ),
    );
  }
}

Output:

┌──────────────────────────────┐ │ Fade Animation │ ├──────────────────────────────┤ │ │ │ BLUE BOX │ │ │ └──────────────────────────────┘
Result: The blue square gradually appears on the screen using a fade animation.

9(a) Fetch data from a REST API

Question: Fetch data from a REST API.

Program:

import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';

void main() => runApp(MyApp());

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: ApiExample()
    );
  }
}

class ApiExample extends StatefulWidget {
  @override
  _ApiExampleState createState() => _ApiExampleState();
}

class _ApiExampleState extends State<ApiExample> {

  String data = 'Loading...';

  @override
  void initState() {
    super.initState();
    fetchData();
  }

  Future<void> fetchData() async {

    final response = await http.get(
      Uri.parse(
        'https://jsonplaceholder.typicode.com/todos/1'
      )
    );

    if (response.statusCode == 200) {

      final jsonData = json.decode(response.body);

      setState(() {
        data = jsonData['title'];
      });

    } else {

      setState(() {
        data = 'Failed to load data';
      });

    }
  }

  @override
  Widget build(BuildContext context) {

    return Scaffold(

      appBar: AppBar(
        title: Text('Fetch API Data')
      ),

      body: Center(
        child: Text(
          data,
          style: TextStyle(fontSize: 20)
        )
      ),

    );
  }
}

Output:

┌──────────────────────────────┐ │ Fetch API Data │ ├──────────────────────────────┤ │ │ │ delectus aut autem │ │ │ └──────────────────────────────┘
Result: Data is successfully fetched from the REST API and displayed in the Flutter application.

9(b) Display the fetched data in a meaningful way in the UI

Question: Display the fetched data in a meaningful way in the UI.

Program:

import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';

void main() => runApp(MyApp());

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: ApiListExample()
    );
  }
}

class ApiListExample extends StatefulWidget {
  @override
  _ApiListExampleState createState() => _ApiListExampleState();
}

class _ApiListExampleState extends State<ApiListExample> {

  List todos = [];
  bool loading = true;

  @override
  void initState() {
    super.initState();
    fetchTodos();
  }

  Future<void> fetchTodos() async {

    final response = await http.get(
      Uri.parse(
        'https://jsonplaceholder.typicode.com/todos'
      )
    );

    if (response.statusCode == 200) {

      setState(() {
        todos = json.decode(response.body);
        loading = false;
      });

    } else {

      setState(() {
        loading = false;
      });

    }
  }

  @override
  Widget build(BuildContext context) {

    return Scaffold(

      appBar: AppBar(
        title: Text('To-Do List')
      ),

      body: loading

          ? Center(
              child: CircularProgressIndicator()
            )

          : ListView.builder(

              itemCount: todos.length,

              itemBuilder: (context, index) {

                return ListTile(

                  title: Text(
                    todos[index]['title']
                  ),

                  trailing: todos[index]['completed']

                      ? Icon(
                          Icons.check,
                          color: Colors.green
                        )

                      : Icon(
                          Icons.close,
                          color: Colors.red
                        ),
                );
              },
            ),
    );
  }
}

Output:

┌────────────────────────────────────┐ │ To-Do List │ ├────────────────────────────────────┤ │ delectus aut autem ✓ │ │ quis ut nam facilis et officia ✗ │ │ fugiat veniam minus ✓ │ │ et porro tempora ✗ │ │ laboriosam mollitia et enim ✓ │ │ ... │ └────────────────────────────────────┘
Result: The REST API data is fetched and displayed as a meaningful To-Do List, with a green check mark for completed tasks and a red cross for incomplete tasks.
