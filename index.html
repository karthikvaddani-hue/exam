void dijkstra(int G[MAX][MAX], int n, int startnode) {
    int cost[MAX][MAX], distance[MAX], pred[MAX];
    int visited[MAX], count, mindistance, nextnode, i, j;

    // Build cost matrix: 0 weights (no edge) become INFINITY
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (G[i][j] == 0)
                cost[i][j] = INFINITY;
            else
                cost[i][j] = G[i][j];
        }
    }

    // Initialize pred[], distance[], and visited[]
    for (i = 0; i < n; i++) {
        distance[i] = cost[startnode][i];
        pred[i] = startnode;
        visited[i] = 0;
    }

    distance[startnode] = 0;
    visited[startnode] = 1;
    count = 1;

    // Main Dijkstra loop
    while (count < n - 1) {
        mindistance = INFINITY;
        nextnode = -1;

        // Find unvisited node with minimum distance
        for (i = 0; i < n; i++) {
            if (!visited[i] && distance[i] < mindistance) {
                mindistance = distance[i];
                nextnode = i;
            }
        }

        // If no reachable node remains, terminate early
        if (nextnode == -1) {
            break;
        }

        visited[nextnode] = 1;

        // Relax edges through nextnode
        for (i = 0; i < n; i++) {
            if (!visited[i]) {
                if (mindistance + cost[nextnode][i] < distance[i]) {
                    distance[i] = mindistance + cost[nextnode][i];
                    pred[i] = nextnode;
                }
            }
        }

        count++;
    }

    // Print shortest distance and path for each node
    for (i = 0; i < n; i++) {
        if (i != startnode) {
            if (distance[i] >= INFINITY) {
                printf("\nDistance of node %d = Unreachable", i);
                printf("\nPath = None\n");
            } else {
                printf("\nDistance of node %d = %d", i, distance[i]);
                printf("\nPath = %d", i);

                j = i;
                do {
                    j = pred[j];
                    printf("<-%d", j);
                } while (j != startnode);

                printf("\n");
            }
        }
    }
}

int main() {
    int windowsize, sent = 0, ack, i, c;

    printf("Enter window size: ");
    if (scanf("%d", &windowsize) != 1 || windowsize <= 0) {
        printf("Invalid window size.\n");
        return 1;
    }

    // Transmit initial window
    for (i = 0; i < windowsize; i++) {
        printf("Frame %d has been transmitted.\n", i);
    }
    sent = windowsize;

    while (1) {
        printf("\nPlease enter your choice:\n");
        printf("1 - For lost data or ACK or delayed ACK\n");
        printf("2 - Exit\n");
        printf("Choice: ");
        scanf("%d", &c);

        if (c == 1) {
            printf("\nEnter the frame number (0 to %d): ", windowsize - 1);
            scanf("%d", &ack);

            if (ack < 0 || ack >= windowsize) {
                printf("Invalid frame number! Must be between 0 and %d.\n", windowsize - 1);
                continue;
            }

            // Retransmit from the corrupted frame to the end of the window
            for (i = ack; i < windowsize; i++) {
                printf("Frame %d has been retransmitted.\n", i);
            }
        } else if (c == 2) {
            printf("Exiting...\n");
            break;
        } else {
            printf("Invalid choice! Enter 1 or 2.\n");
        }
    }

    return 0;
}

Question: Fetch data from a REST API.

#include 

int main() {
    int a[50], b[100];
    int n, i, j = 0, count = 0;

    printf("Enter frame size in bits: ");
    scanf("%d", &n);

    printf("Enter the frame in the form of 0 and 1: ");
    for (i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    for (i = 0; i < n; i++) {
        b[j++] = a[i];

        if (a[i] == 1) {
            count++;
            // After 5 consecutive 1s, stuff a 0
            if (count == 5) {
                b[j++] = 0;
                count = 0; // Reset counter after stuffing
            }
        } else {
            count = 0; // Reset counter if bit is 0
        }
    }

    printf("\nAfter Bit Stuffing:\n");
    printf("01111110 "); // Starting Flag

    for (i = 0; i < j; i++) {
        printf("%d", b[i]);
    }

    printf(" 01111110\n"); // Ending Flag

    return 0;
}

9(b) Display the fetched data in a meaningful way in the UI

Question: Display the fetched data in a meaningful way in the UI.

Program:

import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'dart:convert';

void main() => runApp(MyApp());

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: ApiListExample()
    );
  }
}

class ApiListExample extends StatefulWidget {
  @override
  _ApiListExampleState createState() => _ApiListExampleState();
}

class _ApiListExampleState extends State<ApiListExample> {

  List todos = [];
  bool loading = true;

  @override
  void initState() {
    super.initState();
    fetchTodos();
  }

  Future<void> fetchTodos() async {

    final response = await http.get(
      Uri.parse(
        'https://jsonplaceholder.typicode.com/todos'
      )
    );

    if (response.statusCode == 200) {

      setState(() {
        todos = json.decode(response.body);
        loading = false;
      });

    } else {

      setState(() {
        loading = false;
      });

    }
  }

  @override
  Widget build(BuildContext context) {

    return Scaffold(

      appBar: AppBar(
        title: Text('To-Do List')
      ),

      body: loading

          ? Center(
              child: CircularProgressIndicator()
            )

          : ListView.builder(

              itemCount: todos.length,

              itemBuilder: (context, index) {

                return ListTile(

                  title: Text(
                    todos[index]['title']
                  ),

                  trailing: todos[index]['completed']

                      ? Icon(
                          Icons.check,
                          color: Colors.green
                        )

                      : Icon(
                          Icons.close,
                          color: Colors.red
                        ),
                );
              },
            ),
    );
  }
}
